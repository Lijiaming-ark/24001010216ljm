核心代码

下面是包饺子流程的 async/await 实现，Promise.all 让和面与和馅同时开始，谁慢就等谁。

function delay(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}

async function makeDumplings() {
  // 1. 买原料（顺序）
  await buyIngredients(6000);

  // 2. 和面 + 和馅（并行）
  await Promise.all([
    makeDough(4000),
    makeFilling(5000)
  ]);

  // 3. 包饺子（顺序，需要面和馅都 ready）
  await wrapDumplings(7000);

  // 4. 吃饺子（顺序）
  await eatDumplings(4000);
}
